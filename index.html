window.verHistoricoItem = (rawDesc) => { const descLimpa = Insumos.extractDesc(rawDesc).toLowerCase().trim(); if(!descLimpa) return UI.toast("Nome de insumo inválido."); const panel = document.getElementById('modal-historico-precos'); // INTELIGÊNCIA: Se clicar no relógio do mesmo item já aberto, ele fecha! if(panel.dataset.item === descLimpa && panel.style.opacity === '1') { return window.fecharPainelHistorico(); } let precosCotados = []; let detalhesCotacoes = []; DB.data.processos.forEach(p => { p.req.itens.forEach(item => { const itemDescLimpa = Insumos.extractDesc(item.desc).toLowerCase().trim(); if(itemDescLimpa === descLimpa) { p.propostas.forEach(prop => { const precoFornecedor = prop.itens[item.id]; if(precoFornecedor && precoFornecedor > 0) { precosCotados.push(precoFornecedor); // Captura os detalhes para a lista lateral const fornNome = prop.nome || 'Fornecedor Desconhecido'; const dtCotacao = p.req.emissao || p.criadoEm || Util.today(); detalhesCotacoes.push({ forn: fornNome, data: dtCotacao, preco: precoFornecedor, procId: p.id }); } }); } }); }); if(precosCotados.length === 0) return UI.toast("Ainda não há histórico cotado para este insumo."); const menor = Math.min(...precosCotados); const maior = Math.max(...precosCotados); const media = precosCotados.reduce((a, b) => a + b, 0) / precosCotados.length; // Popula o Lado Esquerdo (Resumo) document.getElementById('hist-insumo-nome').textContent = rawDesc; document.getElementById('hist-menor').textContent = Util.formatBRL(menor); document.getElementById('hist-maior').textContent = Util.formatBRL(maior); document.getElementById('hist-media').textContent = Util.formatBRL(media); document.getElementById('hist-qtd').textContent = precosCotados.length; // Popula o Lado Direito (Lista Detalhada) const listaHtml = document.getElementById('hist-lista-cotacoes'); listaHtml.innerHTML = ''; // Inverte para as mais recentes ficarem no topo da lista detalhesCotacoes.reverse().forEach(c => { listaHtml.innerHTML += `
${c.forn} ${c.data} • Processo: ${c.procId}
${Util.formatBRL(c.preco)}
`; }); panel.dataset.item = descLimpa; panel.style.transform = 'translate(-50%, -50%)'; // Centraliza perfeitamente panel.style.opacity = '1'; panel.style.pointerEvents = 'all'; };
